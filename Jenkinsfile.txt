pipeline {
    agent any

    tools {
        maven 'MAVEN_HOME' // Remplace par le nom exact de ton Maven installé dans Jenkins
    }

    stages {
        stage('Checkout') {
            steps {
                git 'https://github.com/WaelBenOthmane/Saucelabs-Cucumber-Automation.git'
            }
        }

        stage('Build & Test') {
            steps {
                sh 'mvn clean test'
            }
        }

        stage('Cucumber HTML Report') {
            steps {
                publishHTML([
                    reportDir: 'target', 
                    reportFiles: 'cucumber-report.html', 
                    reportName: 'Cucumber Report'
                ])
            }
        }
    }

    post {
        always {
            junit '**/target/surefire-reports/*.xml'
        }
    }
}
